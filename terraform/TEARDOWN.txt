# BEFORE YOU START
# ═══════════════════════════════════════════════════════════════════════════════
#
# 1. AWS CLI on your PC (same account used to deploy).
# 2. Bastion still running and reachable (required for stacks 07–13).
# 3. Expect 20–45 minutes (EKS destroy is slow).
# 4. This is destructive — back up anything you need first.
#
# SSH to bastion (from your PC):
#   cd terraform/06_bastion && terraform output -raw ssh_command

# Per SSH session on bastion:
#   export TF_REGISTRY_CLIENT_TIMEOUT=300
#   export TF_PLUGIN_CACHE_DIR=$HOME/.terraform.d/plugin-cache

# If terraform reports a state lock:
#   terraform force-unlock <LOCK_ID>

# PHASE 1 — DELETE EASYSHOP (bastion)
# ═══════════════════════════════════════════════════════════════════════════════
#
# Check:
#   kubectl get applications -n argocd
#   kubectl get ns easyshop
#
# If the Argo CD application exists:
#   kubectl delete application easyshop -n argocd --wait
#
# If there is no application but the namespace still exists:
#   kubectl delete namespace easyshop --wait
#
# Verify (namespace must be gone):
#   kubectl get ns easyshop
#   kubectl get pvc -A
#
# If namespace is stuck Terminating:
#   kubectl api-resources --verbs=list --namespaced -o name \
#     | xargs -n1 kubectl get -n easyshop 2>/dev/null
